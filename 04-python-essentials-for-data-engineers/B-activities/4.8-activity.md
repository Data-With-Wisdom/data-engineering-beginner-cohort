### **Exercise 1: Safe Division**
```python
def safe_divide(a, b):
    try:
        return a / b
    except ZeroDivisionError:
        print("Cannot divide by zero")
        return None
    except TypeError:
        print("Both inputs must be numbers")
        return None

# Test
print(safe_divide(10, 2))   # 5.0
print(safe_divide(10, 0))   # Cannot divide by zero, None
print(safe_divide(10, "5")) # Both inputs must be numbers, None
```

### **Exercise 2: Safe File Reader**
```python
def read_file_with_fallback(primary_file, backup_file):
    try:
        with open(primary_file, 'r') as file:
            return file.read()
    except FileNotFoundError:
        print(f"{primary_file} not found, trying backup...")
        try:
            with open(backup_file, 'r') as file:
                return file.read()
        except FileNotFoundError:
            print(f"Both files not found")
            return None

# Test
content = read_file_with_fallback('data.csv', 'data_backup.csv')
```

### **Exercise 3: Validate and Convert**
```python
def get_valid_age():
    while True:
        try:
            age_str = input("Enter your age: ")
            age = int(age_str)
            
            if age < 0 or age > 120:
                print("Age must be between 0 and 120")
                continue
            
            return age
        except ValueError:
            print("Please enter a valid number")

# Use it
age = get_valid_age()
print(f"Valid age entered: {age}")
```
