### **Exercise 4: Mask Credit Card**
```python
# Mask all but last 4 digits
def mask_card(card_number):
    # Remove spaces
    card = card_number.replace(" ", "")
    # Mask all but last 4
    masked = "*" * (len(card) - 4) + card[-4:]
    return masked

cards = ["1234 5678 9012 3456", "9876543210123456"]
for card in cards:
    print(mask_card(card))

# Output:
# ************3456
# ************3456
```
